# Absensi Karyawan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Karyawan bisa absen Masuk/Pulang dari HP lewat `/absensi` (selfie + GPS + catatan), dan manajer melihat rekapnya di `/app/laporan/absensi`.

**Architecture:** Halaman karyawan berdiri sendiri di luar `/app` tetapi memakai session Supabase + resolver tenant yang sama. Penulisan data hanya lewat RPC `submit_attendance` (identitas diambil dari `auth.uid()`), sehingga klien tidak bisa memalsukan karyawan/cabang. Rekap adalah halaman `/app` biasa yang terdaftar di nav manifest, memakai helper status murni di TypeScript.

**Tech Stack:** Next.js 16 App Router, React 19, TypeScript, Tailwind CSS 4, Supabase (Postgres + RLS + Storage), vitest, sonner, `GenericExportDialog` (xlsx/PDF).

**Spec:** `docs/superpowers/specs/2026-11-19-absensi-karyawan-design.md`

## Global Constraints

- Semua tanggal & jam dihitung dalam WIB (`Asia/Jakarta`), bukan timezone browser.
- Jam kerja cabang default: `check_in_time = '08:00'`, `check_out_time = '17:00'`.
- Satu baris absen per karyawan per hari: unique `(organization_id, employee_id, attendance_date)`.
- Role yang bisa absen: `MASTER_ADMIN`, `ADMIN`, `FRONTLINER`, `TECHNICIAN`; rekap hanya `MASTER_ADMIN`/`ADMIN`.
- Izin rekap: `reports.attendance` (baru). Menu key: `laporan_absensi`.
- Foto: bucket `cervise-photos`; maksimal 2 MB; `image/jpeg`, `image/png`, `image/webp`; path diawali `attendance/`.
- `/absensi` tetap bisa dibuka walau subscription habis; halaman `/app` lain tidak berubah.
- GPS hanya disimpan; tidak ada validasi radius.
- Bahasa UI: Indonesia, kalimat pendek seperti aplikasi lain.
- ⚠️ Lingkungan saat penulisan: drive C penuh (0 byte), `git commit` gagal. Cek ruang sebelum langkah commit; commit tetap wajib begitu ruang tersedia.

```ts
// src/lib/attendance/status.ts
export type AttendanceStatus =
  | "HADIR" | "TERLAMBAT" | "PULANG_AWAL" | "BELUM_PULANG" | "BELUM_MASUK";

export const ATTENDANCE_STATUS_LABEL: Record<AttendanceStatus, string> = {
  HADIR: "Hadir",
  TERLAMBAT: "Terlambat",
  PULANG_AWAL: "Pulang awal",
  BELUM_PULANG: "Belum pulang",
  BELUM_MASUK: "Belum masuk",
};

const WIB = "Asia/Jakarta";

function asDate(value: Date | string): Date {
  return typeof value === "string" ? new Date(value) : value;
}

export function wibDate(value: Date | string): string {
  const parts = new Intl.DateTimeFormat("en-CA", {
    timeZone: WIB, year: "numeric", month: "2-digit", day: "2-digit",
  }).formatToParts(asDate(value));
  const get = (type: string) => parts.find((p) => p.type === type)?.value ?? "";
  return `${get("year")}-${get("month")}-${get("day")}`;
}

export function wibMinutes(value: Date | string): number {
  const parts = new Intl.DateTimeFormat("en-GB", {
    timeZone: WIB, hour: "2-digit", minute: "2-digit", hourCycle: "h23",
  }).formatToParts(asDate(value));
  const get = (type: string) => Number(parts.find((p) => p.type === type)?.value ?? 0);
  return get("hour") * 60 + get("minute");
}

function timeToMinutes(time: string): number {
  const [hour, minute] = time.split(":").map(Number);
  return (hour || 0) * 60 + (minute || 0);
}

export function attendanceStatus(row: {
  checkInAt: string | null;
  checkOutAt: string | null;
  checkInTime: string;
  checkOutTime: string;
}): AttendanceStatus {
  if (!row.checkInAt) return "BELUM_MASUK";
  if (!row.checkOutAt) return "BELUM_PULANG";
  if (wibMinutes(row.checkInAt) > timeToMinutes(row.checkInTime)) return "TERLAMBAT";
  if (wibMinutes(row.checkOutAt) < timeToMinutes(row.checkOutTime)) return "PULANG_AWAL";
  return "HADIR";
}
```

- [ ] **Step 1: Tulis migrasi lengkap**

```sql
-- Absensi karyawan: satu baris per karyawan per hari (WIB).
create table if not exists public.attendance_records (
  id uuid primary key default gen_random_uuid(),
  organization_id uuid not null references public.organizations(id) on delete cascade,
  branch_id uuid not null references public.branches(id) on delete cascade,
  employee_id uuid not null references public.profiles(id) on delete cascade,
  attendance_date date not null,
  check_in_at timestamptz,
  check_out_at timestamptz,
  check_in_selfie_path text,
  check_out_selfie_path text,
  check_in_lat double precision,
  check_in_lng double precision,
  check_out_lat double precision,
  check_out_lng double precision,
  note text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint attendance_records_one_per_day
    unique (organization_id, employee_id, attendance_date),
  constraint attendance_records_checkout_valid
    check (check_out_at is null or (check_in_at is not null and check_out_at > check_in_at))
);

create index if not exists attendance_records_branch_date_idx
  on public.attendance_records(organization_id, branch_id, attendance_date desc);

alter table public.branches
  add column if not exists check_in_time time not null default '08:00',
  add column if not exists check_out_time time not null default '17:00';

alter table public.attendance_records enable row level security;

-- Urutan OR penting: is_platform_admin() lebih dulu, karena
-- can_access_tenant_branch() adalah SECURITY DEFINER yang membaca employees.
drop policy if exists attendance_records_select on public.attendance_records;
create policy attendance_records_select on public.attendance_records
  for select to authenticated
  using (
    public.is_platform_admin()
    or (
      public.can_access_tenant_branch(branch_id)
      and (
        employee_id = auth.uid()
        or public.can_access_tenant_branch(branch_id, array['MASTER_ADMIN', 'ADMIN']::text[])
      )
    )
  );

-- Tidak ada INSERT/UPDATE/DELETE langsung: semua tulis lewat RPC di bawah.
revoke all on public.attendance_records from anon;
revoke insert, update, delete on public.attendance_records from authenticated;
grant select on public.attendance_records to authenticated;
grant select, insert, update, delete on public.attendance_records to service_role;

create or replace function public.submit_attendance(
  p_kind text,
  p_selfie_path text,
  p_lat double precision,
  p_lng double precision,
  p_note text
) returns public.attendance_records
language plpgsql
security definer
set search_path = public
as $$
declare
  v_employee public.employees%rowtype;
  v_date date := (now() at time zone 'Asia/Jakarta')::date;
  v_row public.attendance_records%rowtype;
begin
  if p_kind not in ('check_in', 'check_out') then
    raise exception 'Jenis absen tidak valid';
  end if;
  if p_selfie_path is null or p_selfie_path = '' then
    raise exception 'Foto selfie wajib diisi';
  end if;

  select * into v_employee
  from public.employees
  where profile_id = auth.uid() and is_active = true;
  if not found then
    raise exception 'Data karyawan tidak ditemukan';
  end if;
  if v_employee.branch_id is null then
    raise exception 'Cabang karyawan belum ditetapkan';
  end if;

  if p_kind = 'check_in' then
    insert into public.attendance_records (
      organization_id, branch_id, employee_id, attendance_date,
      check_in_at, check_in_selfie_path, check_in_lat, check_in_lng, note
    ) values (
      v_employee.organization_id, v_employee.branch_id, v_employee.profile_id, v_date,
      now(), p_selfie_path, p_lat, p_lng, left(coalesce(p_note, ''), 500)
    )
    on conflict (organization_id, employee_id, attendance_date) do nothing
    returning * into v_row;
    if not found then
      raise exception 'Sudah absen masuk hari ini';
    end if;
  else
    update public.attendance_records
       set check_out_at = now(),
           check_out_selfie_path = p_selfie_path,
           check_out_lat = p_lat,
           check_out_lng = p_lng,
           note = coalesce(nullif(left(coalesce(p_note, ''), 500), ''), note),
           updated_at = now()
     where organization_id = v_employee.organization_id
       and employee_id = v_employee.profile_id
       and attendance_date = v_date
       and check_out_at is null
    returning * into v_row;
    if not found then
      raise exception 'Belum absen masuk atau sudah absen pulang hari ini';
    end if;
  end if;

  return v_row;
end;
$$;

revoke all on function public.submit_attendance(text, text, double precision, double precision, text) from public, anon;
grant execute on function public.submit_attendance(text, text, double precision, double precision, text) to authenticated;
grant execute on function public.submit_attendance(text, text, double precision, double precision, text) to service_role;

-- View untuk rekap: menggabungkan nama karyawan, nama cabang, dan jam standar.
-- security_invoker = true supaya RLS tabel tetap dipakai.
create or replace view public.attendance_rekap
  with (security_invoker = true) as
select a.*, p.full_name, b.name as branch_name,
       b.check_in_time, b.check_out_time
from public.attendance_records a
join public.profiles p on p.id = a.employee_id
join public.branches b on b.id = a.branch_id;
grant select on public.attendance_rekap to authenticated;
revoke all on public.attendance_rekap from anon;
```

```ts
// src/app/absensi/actions.ts
"use server";

import { revalidatePath } from "next/cache";
import { getActiveTenantForAttendance } from "@/lib/supabase/actor";
import { createAdminClient } from "@/lib/supabase/admin";
import { PHOTO_BUCKET, attendancePhotoPath, isSafeObjectPath, validatePhotoFile } from "@/lib/photos";
import { wibDate } from "@/lib/attendance/status";

function numberOrNull(value: FormDataEntryValue | null): number | null {
  if (typeof value !== "string" || value.trim() === "") return null;
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : null;
}

export async function submitAttendance(formData: FormData): Promise<{ ok: true }> {
  const kind = formData.get("kind");
  if (kind !== "check_in" && kind !== "check_out") throw new Error("Jenis absen tidak valid");

  const selfie = formData.get("selfie");
  if (!(selfie instanceof File) || selfie.size === 0) throw new Error("Foto selfie wajib");
  const validation = validatePhotoFile(selfie);
  if (!validation.ok) throw new Error(validation.error);

  const actor = await getActiveTenantForAttendance();
  const admin = createAdminClient();
  const path = attendancePhotoPath(
    actor.orgId, actor.employeeId, wibDate(new Date()),
    kind === "check_in" ? "in" : "out", validation.extension,
  );

  const { error: uploadError } = await admin.storage
    .from(PHOTO_BUCKET)
    .upload(path, selfie, { contentType: selfie.type, upsert: false });
  if (uploadError) throw new Error(`Gagal mengunggah foto: ${uploadError.message}`);

  const noteValue = formData.get("note");
  const { error } = await actor.supabase.rpc("submit_attendance", {
    p_kind: kind,
    p_selfie_path: path,
    p_lat: numberOrNull(formData.get("lat")),
    p_lng: numberOrNull(formData.get("lng")),
    p_note: typeof noteValue === "string" ? noteValue : null,
  });
  if (error) {
    // Foto sudah terunggah tapi baris absen gagal: jangan tinggalkan sampah.
    if (isSafeObjectPath(path)) await admin.storage.from(PHOTO_BUCKET).remove([path]);
    throw new Error(error.message);
  }

  revalidatePath("/absensi");
  revalidatePath("/app/laporan/absensi");
  return { ok: true };
}
```

```ts
export type RekapRow = {

type RawRow = {
  id: string; attendance_date: string; employee_id: string; branch_id: string;
  check_in_at: string | null; check_out_at: string | null;
  check_in_selfie_path: string | null; check_out_lat: number | null;
  check_out_lng: number | null; note: string | null;
  full_name: string | null; branch_name: string | null;
  check_in_time: string; check_out_time: string;
};

const time = (value: string | null) =>
  value ? new Intl.DateTimeFormat("id-ID", { timeZone: "Asia/Jakarta", hour: "2-digit", minute: "2-digit" }).format(new Date(value)) : "-";

export function toRekapRow(r: RawRow): RekapRow {
  const status = attendanceStatus({
    checkInAt: r.check_in_at, checkOutAt: r.check_out_at,
    checkInTime: r.check_in_time, checkOutTime: r.check_out_time,
  });
  return {
    id: r.id,
    tanggal: r.attendance_date,
    nama: r.full_name ?? r.employee_id,
    cabang: r.branch_name ?? r.branch_id,
    jamMasuk: time(r.check_in_at),
    jamPulang: time(r.check_out_at),
    status,
    statusLabel: ATTENDANCE_STATUS_LABEL[status],
    selfieUrl: publicPhotoUrl(process.env.NEXT_PUBLIC_SUPABASE_URL ?? "", r.check_in_selfie_path),
    lokasi: r.check_out_lat === null ? "Tidak tersedia" : `${r.check_out_lat}, ${r.check_out_lng}`,
```

```ts
// src/app/app/laporan/absensi/actions.ts
"use server";

import { requireMenu } from "@/lib/auth/require-menu";
import { toRekapRow, type RekapRow } from "./rekap";

export type RekapSummary = {
  employeeId: string; nama: string; hadir: number;
  terlambat: number; pulangAwal: number; belumPulang: number;
};

export async function getAttendanceRekap(
  month: string, branchId: string | null, status: string | null,
): Promise<{ rows: RekapRow[]; summary: RekapSummary[]; branches: { id: string; name: string }[] }> {
  const actor = await requireMenu("laporan_absensi");
  if (!/^\d{4}-\d{2}$/.test(month)) throw new Error("Bulan tidak valid");
  const from = `${month}-01`;
  const [y, m] = month.split("-").map(Number);
  const to = new Date(Date.UTC(y, m, 0)).toISOString().slice(0, 10);

  let query = actor.supabase
    .from("attendance_rekap")
    .select("id,attendance_date,employee_id,branch_id,check_in_at,check_out_at,check_in_selfie_path,check_out_lat,check_out_lng,note,full_name,branch_name,check_in_time,check_out_time")
    .eq("organization_id", actor.orgId)
    .gte("attendance_date", from)
    .lte("attendance_date", to)
    .order("attendance_date", { ascending: false });
  if (branchId) query = query.eq("branch_id", branchId);

  const { data, error } = await query;
  if (error) throw new Error(error.message);

  const rows: RekapRow[] = (data ?? []).map((r: any) => toRekapRow(r));
  const filtered = status ? rows.filter((r) => r.status === status) : rows;

  const summary = new Map<string, RekapSummary>();
  for (const r of rows) {
    const s = summary.get(r.nama) ?? {
      employeeId: r.nama, nama: r.nama, hadir: 0,
      terlambat: 0, pulangAwal: 0, belumPulang: 0,
    };
    if (r.status === "HADIR") s.hadir++;
    if (r.status === "TERLAMBAT") s.terlambat++;
    if (r.status === "PULANG_AWAL") s.pulangAwal++;
    if (r.status === "BELUM_PULANG") s.belumPulang++;
    summary.set(r.nama, s);
  }

  const { data: branches } = await actor.supabase
    .from("branches").select("id,name").eq("organization_id", actor.orgId).order("name");
  return { rows: filtered, summary: [...summary.values()], branches: branches ?? [] };
}
```

```tsx
  const data = await getAttendanceRekap(month, params.cabang ?? null, params.status ?? null);
  return <RekapClient month={month} cabang={params.cabang ?? ""} status={params.status ?? ""} {...data} />;
}
```
